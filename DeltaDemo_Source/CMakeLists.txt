cmake_minimum_required(VERSION 3.16)
project(DeltaDemo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(DEMO_WITH_SDK "Build real controller support (OFF = UI preview only)" ON)
if(DEMO_CROSS_QT)
  include(cmake/BoardQt.cmake)
else()
  find_package(Qt5 5.15 REQUIRED COMPONENTS Core Widgets Network)
endif()
set(DEMO_RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../DeltaDemo_Runtime" CACHE PATH "Demo output root")
if(WIN32)
  set(DEMO_PLATFORM Windows)
else()
  set(DEMO_PLATFORM RK3588)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${DEMO_RUNTIME_DIR}/${DEMO_PLATFORM}")
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${cfg} "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
endforeach()
add_library(demo_config src/Settings.cpp)
target_include_directories(demo_config PUBLIC src)
target_link_libraries(demo_config PUBLIC Qt5::Core)
if(DEMO_WITH_SDK)
  if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
      message(FATAL_ERROR "E450 SDK requires a 64-bit build")
    endif()
    set(DEMO_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows" CACHE PATH "Vendor SDK root")
    set(_header sldmv.h)
    set(_library "${DEMO_SDK_DIR}/lib/sldmv.lib")
    set(_runtime "${DEMO_SDK_DIR}/lib/sldmv.dll")
  else()
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
      message(FATAL_ERROR "M800 requires ARM64. Use DEMO_WITH_SDK=OFF for UI preview on another CPU.")
    endif()
    set(DEMO_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sdk/rk3588" CACHE PATH "Vendor SDK root")
    set(_header sldml.h)
    set(_library "${DEMO_SDK_DIR}/lib/libsldml.so")
    set(_runtime "${_library}")
  endif()
  foreach(path "${DEMO_SDK_DIR}/include/${_header}" "${_library}" "${_runtime}")
    if(NOT EXISTS "${path}")
      message(FATAL_ERROR "Missing vendor SDK: ${path}. See sdk/README.md; UI preview: -DDEMO_WITH_SDK=OFF")
    endif()
  endforeach()
  add_library(demo_motion src/RobotController.cpp)
  target_include_directories(demo_motion PRIVATE "${DEMO_SDK_DIR}/include")
  target_link_libraries(demo_motion PRIVATE "${_library}" Qt5::Network)
  target_compile_definitions(demo_motion PUBLIC DEMO_WITH_SDK=1)
else()
  add_library(demo_motion src/PreviewController.cpp)
endif()
target_link_libraries(demo_motion PUBLIC demo_config)
add_executable(DeltaDemo WIN32 src/main.cpp src/MainWindow.cpp src/MainWindow.h resources/app.qrc)
if(WIN32)
  target_sources(DeltaDemo PRIVATE resources/app.rc)
  set_source_files_properties(resources/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ico")
  target_include_directories(DeltaDemo PRIVATE resources)
endif()
set(DEMO_EXECUTABLE_NAME "DeltaDemo" CACHE STRING "Executable name; override when the previous demo is running")
set_target_properties(DeltaDemo PROPERTIES OUTPUT_NAME "${DEMO_EXECUTABLE_NAME}")
target_link_libraries(DeltaDemo PRIVATE demo_motion Qt5::Widgets)
if(DEMO_WITH_SDK)
  add_custom_command(TARGET DeltaDemo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${_runtime}" "$<TARGET_FILE_DIR:DeltaDemo>")
  if(UNIX)
    set_target_properties(DeltaDemo PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN;$ORIGIN/lib")
  endif()
endif()
if(MSVC)
  add_compile_options(/utf-8)
  foreach(target demo_config demo_motion DeltaDemo)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /wd4819 /wd4828)
  endforeach()
endif()
enable_testing()
add_executable(settings_test tests/settings_test.cpp)
target_link_libraries(settings_test PRIVATE demo_config)
add_test(NAME settings COMMAND settings_test)
add_executable(controller_test tests/controller_test.cpp src/RobotController.cpp)
target_include_directories(controller_test PRIVATE tests/mock)
target_link_libraries(controller_test PRIVATE demo_config Qt5::Network)
add_test(NAME controller COMMAND controller_test)
add_executable(window_test tests/window_test.cpp src/MainWindow.cpp src/RobotController.cpp)
target_include_directories(window_test PRIVATE tests/mock)
target_link_libraries(window_test PRIVATE demo_config Qt5::Widgets Qt5::Network)
if(WIN32)
  add_test(NAME window COMMAND window_test -platform windows)
else()
  add_test(NAME window COMMAND window_test -platform offscreen)
endif()
add_executable(feedback_test tests/feedback_test.cpp src/RobotController.cpp)
target_include_directories(feedback_test PRIVATE tests/mock)
target_compile_definitions(feedback_test PRIVATE DEMO_PULSE_FEEDBACK_TEST=1)
target_link_libraries(feedback_test PRIVATE demo_config Qt5::Network)
add_test(NAME feedback COMMAND feedback_test)
foreach(testTarget settings_test controller_test window_test feedback_test)
  set_target_properties(${testTarget} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")
  foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${testTarget} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/tests")
  endforeach()
  if(WIN32)
    add_custom_command(TARGET ${testTarget} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt5::Core>" "$<TARGET_FILE_DIR:${testTarget}>"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt5::Network>" "$<TARGET_FILE_DIR:${testTarget}>")
  endif()
endforeach()
if(WIN32)
  add_custom_command(TARGET window_test POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt5::Widgets>" "$<TARGET_FILE_DIR:window_test>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt5::Gui>" "$<TARGET_FILE_DIR:window_test>"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:window_test>/platforms"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt5::QWindowsIntegrationPlugin>" "$<TARGET_FILE_DIR:window_test>/platforms")
endif()
